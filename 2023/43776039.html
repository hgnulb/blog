    题目描述

    ✅ 1489. 找到最小生成树里的关键边和伪关键边

    题意分析

    在所有最小生成树中，找出必须出现的关键边，以及可以出现、但并非必须出现的伪关键边，返回原始边编号。

    解法：Kruskal 枚举每条边

    核心思路

    [!blue]
    先求基准最小生成树，再逐边做排除和强制实验。 用 Kruskal 得到最低总权值 base。排除某条边后无法连通，或最低权值变大，说明它是关键边。

    若不是关键边，再强制选择它：先合并端点并计入权值，然后按权值顺序补齐生成树。若总权值仍为 base，就存在包含它的最小生成树，它是伪关键边。

    每次实验重新创建并查集。排序前保存原始编号，最终返回原始编号而非排序后的位置。

    解题步骤

    1. 保存原始边编号并按权值排序。
    2. 执行一次 Kruskal，得到基准权值。
    3. 依次排除每条边，判断是否为关键边。
    4. 对非关键边强制先选，再判断能否达到基准权值。

    示例：三角形的三条边权都是 1，任选两条总权值都是 2。排除任何一条仍能得到 2，强制任何一条也能得到 2，所以三条都是伪关键边，没有关键边。 若某条边是连接两部分的唯一桥边，排除后不连通，它就是关键边。

    代码实现

    class Solution {
    private static final long INF = Long.MAX_VALUE / 4;

    public List<List<Integer>> findCriticalAndPseudoCriticalEdges(int n, int[][] edges) {
        // 额外保存原始边编号，排序位置只用于本次排除或强制选择。
        int[][] sorted = new int[edges.length][4];

        for (int i = 0; i < edges.length; i++) {
            sorted[i] = new int[] {edges[i][0], edges[i][1], edges[i][2], i};
        }

        Arrays.sort(sorted, (a, b) -> Integer.compare(a[2], b[2]));

        long base = buildMst(n, sorted, -1, -1);
        List<Integer> critical = new ArrayList<>();
        List<Integer> pseudoCritical = new ArrayList<>();

        for (int i = 0; i < sorted.length; i++) {
            // 先判断排除是否变差，再判断非关键边能否参与最优树。
            if (buildMst(n, sorted, i, -1) > base) {
                critical.add(sorted[i][3]);
            } else if (buildMst(n, sorted, -1, i) == base) {
                pseudoCritical.add(sorted[i][3]);
            }
        }

        return Arrays.asList(critical, pseudoCritical);
    }

    private long buildMst(int n, int[][] edges, int skip, int force) {
        // 每轮独立构造并查集，不能继承其他实验的合并状态。
        UnionFind unionFind = new UnionFind(n);
        long weight = 0;
        int used = 0;

        // 强制边先合并并计费，再用其余边补齐生成树。
        if (force >= 0) {
            int[] edge = edges[force];

            if (!unionFind.union(edge[0], edge[1])) {
                return INF;
            }

            weight += edge[2];
            used++;
        }

        for (int i = 0; i < edges.length && used < n - 1; i++) {
            if (i == skip || i == force) {
                continue;
            }

            int[] edge = edges[i];

            if (unionFind.union(edge[0], edge[1])) {
                weight += edge[2];
                used++;
            }
        }

        // 不足所需边数表示不连通，不能把部分权值当作生成树费用。
        return used == n - 1 ? weight : INF;
    }

    private static class UnionFind {
        private final int[] parent;
        private final int[] rank;

        UnionFind(int n) {
            parent = new int[n];
            rank = new int[n];

            for (int i = 0; i < n; i++) {
                parent[i] = i;
            }
        }

        int find(int x) {
            while (x != parent[x]) {
                parent[x] = parent[parent[x]];
                x = parent[x];
            }

            return x;
        }

        boolean union(int x, int y) {
            int rootX = find(x);
            int rootY = find(y);

            if (rootX == rootY) {
                return false;
            }

            if (rank[rootX] < rank[rootY]) {
                parent[rootX] = rootY;
            } else if (rank[rootX] > rank[rootY]) {
                parent[rootY] = rootX;
            } else {
                parent[rootY] = rootX;
                rank[rootX]++;
            }

            return true;
        }
    }
}

    import "sort"

func findCriticalAndPseudoCriticalEdges(n int, edges [][]int) [][]int {
    // 额外保存原始边编号，排序位置只用于本次排除或强制选择。
    sorted := make([][]int, len(edges))
    for i, edge := range edges {
        sorted[i] = []int{edge[0], edge[1], edge[2], i}
    }
    sort.Slice(sorted, func(i, j int) bool {
        return sorted[i][2] < sorted[j][2]
    })

    const inf int64 = 1 << 60
    buildMst := func(skip, force int) int64 {
        // 每轮独立构造并查集，不能继承其他实验的合并状态。
        parent := make([]int, n)
        rank := make([]int, n)
        for i := range parent {
            parent[i] = i
        }

        find := func(x int) int {
            for x != parent[x] {
                parent[x] = parent[parent[x]]
                x = parent[x]
            }
            return x
        }
        union := func(x, y int) bool {
            rootX, rootY := find(x), find(y)
            if rootX == rootY {
                return false
            }
            if rank[rootX] < rank[rootY] {
                parent[rootX] = rootY
            } else if rank[rootX] > rank[rootY] {
                parent[rootY] = rootX
            } else {
                parent[rootY] = rootX
                rank[rootX]++
            }
            return true
        }

        var weight int64
        used := 0
        // 强制边先合并并计费，再用其余边补齐生成树。
        if force >= 0 {
            edge := sorted[force]
            if !union(edge[0], edge[1]) {
                return inf
            }
            weight += int64(edge[2])
            used++
        }

        for i, edge := range sorted {
            if used == n-1 {
                break
            }
            if i == skip || i == force {
                continue
            }
            if union(edge[0], edge[1]) {
                weight += int64(edge[2])
                used++
            }
        }
        // 不足所需边数表示不连通，不能把部分权值当作生成树费用。
        if used != n-1 {
            return inf
        }
        return weight
    }

    base := buildMst(-1, -1)
    critical := make([]int, 0)
    pseudoCritical := make([]int, 0)
    for i, edge := range sorted {
        // 先判断排除是否变差，再判断非关键边能否参与最优树。
        if buildMst(i, -1) > base {
            critical = append(critical, edge[3])
        } else if buildMst(-1, i) == base {
            pseudoCritical = append(pseudoCritical, edge[3])
        }
    }
    return [][]int{critical, pseudoCritical}
}

    复杂度分析

    • 时间复杂度：$O(E\log(E+1)+EV+E²α(V))$，排序一次，每条边至多进行两次 Kruskal；每轮还需初始化 V 个并查集节点。
    • 空间复杂度：$O(E+V)$，保存边、编号和并查集。

    关键点总结

    [!green]

    • 排除变差说明必须选，强制不变说明可以选。
    • 先判断关键边，避免再加入伪关键边集合。
    • 生成树必须有 V-1 次成功合并，未连通不能只比较已有权值。

    易错点总结

    [!yellow]

    • 只看某一次生成树是否选了该边：无法区分必选和可替换。
    • 实验之间复用并查集状态：会带入上一次合并。
    • 排序后直接返回当前位置：不符合原始编号要求。
    • 强制边未先合并端点：后续选边可能产生环。
